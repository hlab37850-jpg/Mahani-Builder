function normalizeResumeText(value) {
  return String(value || '')
    .trim()
    .toLowerCase()
    .replace(/\s+/g, ' ');
}

function findResumableProject(appName, idea, appType, design) {
  const projectsRoot = path.join(__dirname, 'projects');

  if (!fs.existsSync(projectsRoot)) {
    return null;
  }

  const target = {
    appName: normalizeResumeText(appName),
    description: normalizeResumeText(idea),
    appType: normalizeResumeText(appType || 'تطبيق عام'),
    design: normalizeResumeText(
      design || 'تصميم احترافي حديث، عربي RTL'
    )
  };

  const candidates = [];

  for (const name of fs.readdirSync(projectsRoot)) {
    const projectRoot = path.join(projectsRoot, name);
    const checkpointPath = path.join(
      projectRoot,
      '.mahani-checkpoint.json'
    );

    if (!fs.existsSync(checkpointPath)) {
      continue;
    }

    try {
      const checkpoint = JSON.parse(
        fs.readFileSync(checkpointPath, 'utf8')
      );

      if (
        !checkpoint ||
        checkpoint.version !== 2 ||
        !checkpoint.projectName ||
        !Array.isArray(checkpoint.completedStages) ||
        checkpoint.completedStages.length === 0 ||
        checkpoint.completedStages.length >= 4
      ) {
        continue;
      }

      const sameApp =
        normalizeResumeText(checkpoint.appName) ===
        target.appName;

      const sameDescription =
        normalizeResumeText(checkpoint.description) ===
        target.description;

      const sameAppType =
        normalizeResumeText(checkpoint.appType) ===
        target.appType;

      const sameDesign =
        normalizeResumeText(checkpoint.design) ===
        target.design;

      if (
        sameApp &&
        sameDescription &&
        sameAppType &&
        sameDesign
      ) {
        candidates.push({
          projectName: checkpoint.projectName,
          completedStages: checkpoint.completedStages,
          mtime: fs.statSync(checkpointPath).mtimeMs
        });
      }
    } catch (error) {
      console.log(
        '[CHECKPOINT] تجاهل checkpoint غير صالح: ' +
        checkpointPath
      );
    }
  }

  if (!candidates.length) {
    console.log(
      '[CHECKPOINT] لا يوجد مشروع مطابق لهذا الطلب'
    );
    return null;
  }

  candidates.sort((a, b) => b.mtime - a.mtime);

  const selected = candidates[0];

  console.log(
    '[CHECKPOINT] تم العثور على مشروع مطابق: ' +
    selected.projectName
  );

  console.log(
    '[CHECKPOINT] المراحل المكتملة: ' +
    selected.completedStages.join(', ')
  );

  return selected.projectName;
}

app.get('/api/project/:name', (req, res) => {
  try {
    const projectName = safeProjectName(req.params.name);

    const projectRoot = path.join(
      __dirname,
      'projects',
      projectName
    );

    if (!fs.existsSync(projectRoot)) {
      return res.status(404).json({
        success: false,
        error: 'المشروع غير موجود'
      });
    }

    const files = [];

    function walk(dir) {
      for (const entry of fs.readdirSync(dir, {
        withFileTypes: true
      })) {
        const full = path.join(dir, entry.name);

        if (entry.isDirectory()) {
          walk(full);
        } else if (entry.isFile()) {
          files.push(
            path.relative(projectRoot, full)
          );
        }
      }
    }

    walk(projectRoot);

    res.json({
      success: true,
      project: projectName,
      files: files.sort()
    });

  } catch (error) {
    res.status(500).json({
      success: false,
      error: error.message
    });
  }
});

function safeProjectName(name) {
  return String(name || 'mahani_app')
    .trim()
    .toLowerCase()
    .replace(/[^a-z0-9_]+/g, '_')
    .replace(/^_+|_+$/g, '')
    .replace(/_+/g, '_')
    .slice(0, 50) || `mahani_app_${Date.now()}`;
}

function root(projectName) {
  return path.join(
    __dirname,
    'projects',
    safeProjectName(projectName)
  );
}


function createFlutterAndroidProject(projectName, plan) {
  const projectRoot = root(projectName);
  const projectNameSafe = safeProjectName(projectName);

  fs.mkdirSync(
    path.dirname(projectRoot),
    { recursive: true }
  );

  console.log(
    '[ENGINE] إنشاء Flutter Android project حقيقي'
  );

  if (
    fs.existsSync(projectRoot) &&
    fs.existsSync(path.join(projectRoot, 'android'))
  ) {
    console.log(
      '[ENGINE] Android project موجود بالفعل'
    );
    return projectRoot;
  }

  if (fs.existsSync(projectRoot)) {
    fs.rmSync(
      projectRoot,
      {
        recursive: true,
        force: true
      }
    );
  }

  execFileSync(
    'flutter',
    [
      'create',
      '--platforms=android',
      '--project-name',
      projectNameSafe,
      projectRoot
    ],
    {
      stdio: 'inherit',
      timeout: 180000
    }
  );

  // Normalize Dart SDK constraint so the generated project
  // remains compatible with the Flutter version used by GitHub Actions.
  const generatedPubspec = path.join(
    projectRoot,
    'pubspec.yaml'
  );

  if (fs.existsSync(generatedPubspec)) {
    let pubspecContent = fs.readFileSync(
      generatedPubspec,
      'utf8'
    );

    pubspecContent = pubspecContent.replace(
      /^\s*sdk:\s*\^3\.\d+\.\d+\s*$/m,
      "    sdk: '>=3.9.0 <4.0.0'"
    );

    fs.writeFileSync(
      generatedPubspec,
      pubspecContent,
      'utf8'
    );

    console.log(
      '[ENGINE] Dart SDK constraint normalized to >=3.9.0 <4.0.0'
    );
  }

  if (
    !fs.existsSync(
      path.join(projectRoot, 'android')
    )
  ) {
    throw new Error(
      'FLUTTER_CREATE_FAILED: مجلد android لم يتم إنشاؤه'
    );
  }

  if (
    !fs.existsSync(
      path.join(projectRoot, 'pubspec.yaml')
    )
  ) {
    throw new Error(
      'FLUTTER_CREATE_FAILED: pubspec.yaml لم يتم إنشاؤه'
    );
  }

  console.log(
    '[ENGINE] تم إنشاء Flutter Android project بنجاح'
  );

  return projectRoot;
}

async function askAI(apiKey, messages, maxTokens = 5000) {
  const { execFile } = require('child_process');

  const models = [
    'nvidia/nemotron-3-super-120b-a12b:free',
    ...(process.env.GROQ_API_KEY ? ['groq/openai/gpt-oss-20b'] : []),
    'openai/gpt-oss-20b:free',
    'cohere/north-mini-code:free'
  ].filter((value, index, array) =>
    value && array.indexOf(value) === index
  );

  const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));

  function extractJson(text) {
    let clean = String(text || '')
      .replace(/^```json\s*/i, '')
      .replace(/^```\s*/i, '')
      .replace(/\s*```$/i, '')
      .trim();

    try {
      return JSON.parse(clean);
    } catch (_) {}

    const first = clean.indexOf('{');
    const last = clean.lastIndexOf('}');

    if (first >= 0 && last > first) {
      const candidate = clean.slice(first, last + 1);
      try {
        return JSON.parse(candidate);
      } catch (_) {}
    }

    throw new Error(
      'AI returned invalid/incomplete JSON: ' +
      clean.slice(0, 3000)
    );
  }

  function callCurl(model, tokenLimit) {
    const isGroq = model.startsWith('groq/');
    const provider = isGroq ? 'Groq' : 'OpenRouter';

    const actualModel = isGroq
      ? model.slice(5)
      : model;

    const endpoint = isGroq
      ? 'https://api.groq.com/openai/v1/chat/completions'
      : 'https://openrouter.ai/api/v1/chat/completions';

    const providerKey = isGroq
      ? process.env.GROQ_API_KEY
      : apiKey;

    if (!providerKey) {
      throw new Error(provider + ' API key is missing');
    }

    const payloadObject = {
      model: actualModel,
      messages,
      temperature: 0,
      response_format: { type: 'json_object' }
    };

    if (isGroq) {
      payloadObject.max_completion_tokens = tokenLimit;
    } else {
      payloadObject.max_tokens = tokenLimit;
    }

    const payload = JSON.stringify(payloadObject);

    console.log(
      '[AI] provider=' + provider +
      ' model=' + actualModel +
      ' tokens=' + tokenLimit
    );

    return new Promise((resolve, reject) => {
      execFile(
        'curl',
        [
          '--silent',
          '--show-error',
          '--fail-with-body',
          '--connect-timeout', '30',
          '--max-time', '120',
          endpoint,
          '-H', 'Authorization: Bearer ' + providerKey,
          '-H', 'Content-Type: application/json',

          ...(isGroq
            ? []
            : [
                '-H', 'HTTP-Referer: http://127.0.0.1:3000',
                '-H', 'X-Title: Mahani Builder AI'
              ]),

          '--data', payload
        ],
        {
          maxBuffer: 30 * 1024 * 1024
        },
        (error, stdout, stderr) => {
          const raw = String(stdout || '').trim();
          const err = String(stderr || '').trim();

          let data = null;

          if (raw) {
            try {
              data = JSON.parse(raw);
            } catch (_) {}
          }

          if (error) {
            const providerError =
              data && data.error
                ? JSON.stringify(data.error)
                : raw.slice(0, 5000);

            reject(new Error(
              provider + ' curl failed: ' +
              (err || error.message) +
              '; RESPONSE=' +
              providerError
            ));
            return;
          }

          if (!data) {
            reject(new Error(
              provider + ' returned empty JSON'
            ));
            return;
          }

          if (data.error) {
            reject(new Error(
              provider + ' API error: ' +
              JSON.stringify(data.error).slice(0, 5000)
            ));
            return;
          }

          const choice = data.choices?.[0];
          const finish =
            choice?.finish_reason || 'unknown';

          const content =
            choice?.message?.content;

          console.log(
            '[AI] model=' +
            (data.model || actualModel) +
            ' provider=' +
            (data.provider || provider) +
            ' tokens=' +
            tokenLimit +
            ' finish=' +
            finish +
            ' content=' +
            (typeof content === 'string'
              ? content.length
              : 0)
          );

          if (!content) {
            reject(new Error(
              'AI returned no content: finish_reason=' +
              finish +
              '; usage=' +
              JSON.stringify(
                data.usage || {}
              ).slice(0, 3000)
            ));
            return;
          }

          try {
            resolve(extractJson(content));
          } catch (jsonError) {
            reject(new Error(
              jsonError.message +
              '; finish_reason=' +
              finish
            ));
          }
        }
      );
    });
  }

  const firstLimit = Math.min(
    Math.max(2500, Number(maxTokens) || 6000),
    9000
  );

  let lastError = null;

  for (const model of models) {
    console.log('');
    console.log('[AI] محاولة النموذج: ' + model);

    try {
      console.log('[AI] tokens=' + firstLimit);
      return await callCurl(model, firstLimit);

    } catch (firstError) {
      lastError = firstError;

      const msg = String(firstError.message || '');

      console.log(
        '[AI] فشل النموذج ' + model + ':'
      );
      console.log(msg);

      // 401: توقف فوريًا لأن المفتاح نفسه غير صالح.
      if (/401|invalid.*api.*key|authentication/i.test(msg)) {
        throw new Error(
          'OPENROUTER_AUTH_FAILED: ' + msg
        );
      }

      // 429: لا تعيد المحاولة بنفس النموذج.
      // وإذا كان الحد اليومي عامًا، لا تحرق بقية الطلبات.
      if (
        /429|free-models-per-day|rate.?limit|too many requests/i.test(msg)
      ) {
        console.log(
          '[AI] 429 — تم إيقاف إعادة المحاولة لهذا النموذج'
        );

        if (/free-models-per-day/i.test(msg)) {
          lastError = new Error(
            'OPENROUTER_FREE_DAILY_LIMIT_REACHED'
          );
          continue;
        }

        await sleep(1500);
        continue;
      }

      // 402 / credits / quota
      if (/402|payment|insufficient|credits|quota/i.test(msg)) {
        console.log(
          '[AI] الرصيد/الحصة غير متاحة — النموذج التالي'
        );
        continue;
      }

      // Timeout / 5xx:
      // تجربة واحدة فقط بحجم أصغر.
      if (
        /500|502|503|504|timeout|timed out|curl failed/i.test(msg)
      ) {
        console.log(
          '[AI] خطأ مؤقت — محاولة أصغر مرة واحدة'
        );

        try {
          return await callCurl(
            model,
            Math.max(3000, Math.floor(firstLimit * 0.55))
          );
        } catch (secondError) {
          lastError = secondError;
          console.log(
            '[AI] فشلت المحاولة المختصرة — النموذج التالي'
          );
          continue;
        }
      }

      // JSON غير صالح أو إخراج مبتور:
      // لا نكرر 8000 token؛ ننتقل للنموذج التالي.
      if (
        /invalid.*json|incomplete.*json|finish_reason=length/i.test(msg)
      ) {
        console.log(
          '[AI] JSON غير صالح/مبتور — النموذج التالي'
        );
        continue;
      }

      continue;
    }
  }

  if (
    lastError &&
    lastError.message === 'OPENROUTER_FREE_DAILY_LIMIT_REACHED'
  ) {
    throw new Error(
      'OPENROUTER_FREE_DAILY_LIMIT_REACHED: ' +
      'تم استنفاد حد OpenRouter المجاني اليومي. ' +
      'لن يتم حرق طلبات إضافية بإعادة المحاولة.'
    );
  }

  throw new Error(
    'AI generation failed: all fallback models failed. LAST=' +
    (lastError ? lastError.message : 'unknown error')
  );
}

function collectDartContext(projectName, maxFiles = 40, maxCharsPerFile = 14000, maxTotalChars = 90000) {
  const projectRoot = root(projectName);
  const files = collectFiles(projectName);
  const result = [];
  let total = 0;

  for (const relative of files.slice(0, maxFiles)) {
    const full = path.join(projectRoot, relative);

    if (!fs.existsSync(full)) {
      continue;
    }

    let content = fs.readFileSync(full, 'utf8');

    if (content.length > maxCharsPerFile) {
      content =
        content.slice(0, maxCharsPerFile) +
        '\n\n// [CONTEXT TRUNCATED]\n';
    }

    if (total + content.length > maxTotalChars) {
      break;
    }

    result.push({
      path: relative,
      content
    });

    total += content.length;
  }

  return result;
}

function generateStage(
  apiKey,
  plan,
  stage,
  existingFiles
) {
  const projectName = safeProjectName(
    plan?.appName || plan?.name || 'mahani_app'
  );
  const stageLimits = {
    core: 1,
    data: 2,
    services: 2,
    ui: 3
  };

  const maxFiles = stageLimits[stage] || 2;

  // Build a real project context from the current Dart files.
  // This gives every generation stage access to actual source code,
  // not only a list of filenames.
  const contextMaxFiles =
      stage === 'core' ? 8 :
      stage === 'ui' ? 8 : 24;

    const contextMaxCharsPerFile =
      stage === 'core' ? 6000 :
      stage === 'ui' ? 4000 : 9000;

    const contextMaxTotalChars =
      stage === 'core' ? 18000 :
      stage === 'ui' ? 18000 : 60000;

    const projectContext = collectDartContext(
      projectName,
      contextMaxFiles,
      contextMaxCharsPerFile,
      contextMaxTotalChars
    );

  const compactPrompt = `
You are a senior Flutter production engineer working inside Mahani Builder AI.

Generate ONLY the current stage of a real Flutter application.

PROJECT:
${JSON.stringify(plan)}

function normalizeLocalPackageImports(projectName, source) {
  const projectPackage = safeProjectName(projectName);

  return String(source || '').replace(
    /package:([a-zA-Z0-9_]+)\//g,
    (full, packageName) => {
      // Keep the current project's own package imports unchanged.
      if (packageName === projectPackage) {
        return full;
      }

      // Keep every known external Flutter/Dart package unchanged.
      if (PACKAGE_VERSIONS[packageName]) {
        return full;
      }

      // Flutter SDK packages must ALWAYS remain external imports.
      if (
        packageName === 'flutter' ||
        packageName === 'flutter_localizations' ||
        packageName === 'cupertino'
      ) {
        return full;
      }

      // Unknown package imports are left untouched.
      // The validator will report them if the dependency is missing.
      return full;
    }
  );
}

function writeFiles(projectName, files) {
  const projectRoot = root(projectName);
  const libRoot = path.join(
    projectRoot,
    'lib'
  );

  fs.mkdirSync(
    libRoot,
    { recursive: true }
  );

  let written = 0;

  for (
    const file of Array.isArray(files)
      ? files
      : []
  ) {
    if (
      !file ||
      typeof file !== 'object'
    ) {
      continue;
    }

    const relative = String(
      file.path || ''
    ).trim();

    if (
      !relative.startsWith('lib/') ||
      relative.includes('..') ||
      relative.includes('\\') ||
      !relative.endsWith('.dart')
    ) {
      continue;
    }

    const full = path.resolve(
      projectRoot,
      relative
    );

    if (
      !full.startsWith(
        path.resolve(libRoot) +
        path.sep
      )
    ) {
      continue;
    }

    const content = normalizeLocalPackageImports(
      projectName,
      String(file.content || '').trim()
    );

    if (!content) continue;

    fs.mkdirSync(
      path.dirname(full),
      { recursive: true }
    );

    fs.writeFileSync(
      full,
      content + '\n',
      'utf8'
    );

    written++;
  }

  return written;
}

function collectFiles(projectName) {
  const projectRoot = root(projectName);
  const result = [];

  if (!fs.existsSync(projectRoot)) {
    return result;
  }

  function walk(dir) {
    for (
      const item of fs.readdirSync(
        dir,
        { withFileTypes: true }
      )
    ) {
      const full = path.join(
        dir,
        item.name
      );

      if (item.isDirectory()) {
        walk(full);
      } else if (
        item.isFile() &&
        item.name.endsWith('.dart')
      ) {
        result.push(
          path.relative(
            projectRoot,
            full
          )
        );
      }
    }
  }

  walk(
    path.join(
      projectRoot,
      'lib'
    )
  );

  return result.sort();
}

function collectAllProjectFiles(projectName) {
  const projectRoot = root(projectName);

  if (!fs.existsSync(projectRoot)) {
    return [];
  }

  const result = [];

  function walk(dir) {
    for (
      const item of fs.readdirSync(
        dir,
        { withFileTypes: true }
      )
    ) {
      const full = path.join(
        dir,
        item.name
      );

      if (item.isDirectory()) {
        walk(full);
      } else {
        result.push(
          path.relative(
            projectRoot,
            full
          )
        );
      }
    }
  }

  walk(projectRoot);

  return result.sort();
}

function detectImportedPackages(
  projectName
) {
  const projectRoot = root(projectName);
  const packages = new Set();

  for (
    const relative of collectFiles(projectName)
  ) {
    const full = path.join(
      projectRoot,
      relative
    );

    const source = fs.readFileSync(
      full,
      'utf8'
    );

    const regex =
      /import\s+['"]package:([^/]+)\//g;

    let match;

    while (
      (match = regex.exec(source))
    ) {
      const packageName = match[1];

      if (
        PACKAGE_VERSIONS[packageName]
      ) {
        packages.add(packageName);
      }
    }
  }

  return [...packages].sort();
}

function validateGeneratedFiles(
  projectName
) {
  const projectRoot = root(projectName);

  const errors = [];

  const files =
    collectFiles(projectName);

  if (!files.includes('lib/main.dart')) {
    errors.push(
      'lib/main.dart غير موجود'
    );
  }

  const pubspecPath = path.join(
    projectRoot,
    'pubspec.yaml'
  );

  if (!fs.existsSync(pubspecPath)) {
    errors.push(
      'pubspec.yaml غير موجود'
    );
  }

  const importedPackages =
    detectImportedPackages(projectName);

  for (
    const packageName of importedPackages
  ) {
    if (
      !PACKAGE_VERSIONS[packageName]
    ) {
      errors.push(
        `Package غير معروف: ${packageName}`
      );
    }
  }

  return {
    valid: errors.length === 0,
    errors,
    files,
    importedPackages
  };
}


function createPubspec(projectName, plan) {
  const projectRoot = root(projectName);
  const pubspecPath = path.join(projectRoot, 'pubspec.yaml');

  fs.mkdirSync(projectRoot, { recursive: true });
  fs.mkdirSync(path.join(projectRoot, 'lib'), { recursive: true });

  const dependencies = Object.entries(PACKAGE_VERSIONS)
    .map(([name, version]) => '  ' + name + ': ' + version)
    .join('\n');

  const description = String(
    plan?.idea || plan?.description || 'Flutter application'
  )
    .replace(/\r?\n/g, ' ')
    .replace(/"/g, '\\"')

function ensureFlutterPubspecDependencies(projectName, plan) {
  const projectRoot = root(projectName);

  const pubspecPath = path.join(
    projectRoot,
    'pubspec.yaml'
  );

  if (!fs.existsSync(pubspecPath)) {
    throw new Error(
      'PUBSPEC_NOT_FOUND: Flutter project did not create pubspec.yaml'
    );
  }

  let content = fs.readFileSync(
    pubspecPath,
    'utf8'
  );

  const dependencies = {
    flutter_localizations: null,
    ...PACKAGE_VERSIONS
  };

  const lines = content.split(/\r?\n/);

  let dependenciesIndex = lines.findIndex(
    line => line.trim() === 'dependencies:'
  );

  if (dependenciesIndex === -1) {
    throw new Error(
      'PUBSPEC_INVALID: dependencies section not found'
    );
  }

  let dependenciesEnd = lines.findIndex(
    (line, index) =>
      index > dependenciesIndex &&
      line.length > 0 &&
      !line.startsWith(' ') &&
      !line.startsWith('#')
  );

  if (dependenciesEnd === -1) {
    dependenciesEnd = lines.length;
  }

  const existing = new Set();

  for (
    let i = dependenciesIndex + 1;
    i < dependenciesEnd;
    i++
  ) {
    const match = lines[i].match(
      /^  ([a-zA-Z0-9_]+):/
    );

    if (match) {
      existing.add(match[1]);
    }
  }

  const additions = [];

  if (!existing.has('flutter_localizations')) {
    additions.push(
      '  flutter_localizations:',
      '    sdk: flutter'
    );
  }

  for (const [name, version] of Object.entries(
    PACKAGE_VERSIONS
  )) {
    if (existing.has(name)) {
      continue;
    }

    additions.push(
      '  ' + name + ': ' + version
    );
  }

  if (additions.length > 0) {
    lines.splice(
      dependenciesEnd,
      0,
      ...additions
    );
  }

  content = lines.join('\n');

  fs.writeFileSync(
    pubspecPath,
    content,
    'utf8'
  );

  console.log(
    '[ENGINE] Flutter pubspec dependencies configured'
  );

  return pubspecPath;
}

function validateDartConsistency(projectName) {
  const projectRoot = root(projectName);
  const libRoot = path.join(projectRoot, 'lib');
  const errors = [];

  if (!fs.existsSync(libRoot)) {
    return { valid: false, errors: ['lib directory غير موجود'] };
  }

  const dartFiles = [];

  function walk(dir) {
    for (const item of fs.readdirSync(dir, { withFileTypes: true })) {
      const full = path.join(dir, item.name);

      if (item.isDirectory()) {
        walk(full);
      } else if (item.isFile() && item.name.endsWith('.dart')) {
        dartFiles.push(full);
      }
    }
  }

  walk(libRoot);

  const relativeDartFiles = new Set(
    dartFiles.map(full =>
      path.relative(libRoot, full).split(path.sep).join('/')
    )
  );

  let packageName = safeProjectName(projectName);

  const pubspecPath = path.join(projectRoot, 'pubspec.yaml');

  if (fs.existsSync(pubspecPath)) {
    const pubspec = fs.readFileSync(pubspecPath, 'utf8');
    const match = pubspec.match(/^name:\s*([A-Za-z0-9_]+)\s*$/m);

    if (match) {
      packageName = match[1];
    }
  }

  const declarations = new Map();
  const allSource = new Map();

  for (const full of dartFiles) {
    const relative = path.relative(libRoot, full)
      .split(path.sep)
      .join('/');

    const source = fs.readFileSync(full, 'utf8');

    allSource.set(relative, source);

    const names = [
      ...source.matchAll(
        /\b(?:class|enum|mixin|extension)\s+([A-Za-z_]\w*)/g
      )
    ].map(m => m[1]);

    for (const name of names) {
      if (!declarations.has(name)) {
        declarations.set(name, []);
      }

      declarations.get(name).push(relative);
    }
  }

  for (const [relative, source] of allSource) {
    const imports = [
      ...source.matchAll(/import\s+['"]([^'"]+)['"]/g)
    ].map(m => m[1]);

    for (const importPath of imports) {

      if (importPath.startsWith('package:')) {
        const match = importPath.match(/^package:([^/]+)\/(.+)$/);

        if (!match) {
          errors.push(
            relative + ': package import غير صالح: ' + importPath
          );
          continue;
        }

        const importedPackage = match[1];
        const importedFile = match[2];

        if (importedPackage === packageName) {
          if (!relativeDartFiles.has(importedFile)) {
            errors.push(
              relative +
              ': import داخلي غير موجود: ' +
              importPath
            );
          }
        } else if (!PACKAGE_VERSIONS[importedPackage]) {
          errors.push(
            relative +
            ': package غير موجود في PACKAGE_VERSIONS: ' +
            importedPackage
          );
        }
      }

      if (
        importPath.startsWith('../') ||
        importPath.startsWith('./')
      ) {
        const currentDir = path.posix.dirname(relative);

        const normalized = path.posix.normalize(
          path.posix.join(currentDir, importPath)
        );

        if (!relativeDartFiles.has(normalized)) {
          errors.push(
            relative +
            ': relative import غير موجود: ' +
            importPath
          );
        }
      }
    }
  }

  // Generic project consistency validation.
  // يتم الاعتماد على:
  // 1. صحة مسارات Dart الداخلية.
  // 2. وجود الملفات المستوردة.
  // 3. وجود الحزم المعروفة.
  // 4. وجود lib/main.dart.
  // 5. سلامة اسم package في pubspec.
  const uniqueErrors = [...new Set(errors)];

  return {
    valid: uniqueErrors.length === 0,
    errors: uniqueErrors,
    dartFiles: relativeDartFiles.size,
    packageName
  };
}

            find build -type f -name "*.apk" -print || true
            exit 1
          fi

          echo "APK CREATED SUCCESSFULLY"
          ls -lh "$APK"

      - name: Upload APK
        uses: actions/upload-artifact@v4
        with:
          name: mahani_app-apk
          path: build/app/outputs/flutter-apk/app-release.apk
          if-no-files-found: error
          retention-days: 7
`;

  fs.writeFileSync(
    workflowPath,
    workflow,
    'utf8'
  );

  console.log(
    '[ENGINE] GitHub Actions workflow created: ' +
    workflowPath
  );

  return workflowPath;
}

async function buildProject(
  apiKey,
  projectName,
  plan
) {
  if (!apiKey) {
    throw new Error(
      'OPENROUTER_API_KEY غير موجود'
    );
  }

  const projectRoot =
    createFlutterAndroidProject(
      projectName,
      plan
    );

  ensureFlutterPubspecDependencies(
    projectName,
    plan
  );

  console.log(
    '[ENGINE] Flutter Android project + dependencies configured'
  );

  const stages = [
    'core',
    'data',
    'services',
    'ui'
  ];

    // ============================================================
    // MAHANI CHECKPOINT / RESUME
    // ============================================================

    const checkpointPath = path.join(
      projectRoot,
      '.mahani-checkpoint.json'
    );

    let checkpoint = {
      version: 2,
      projectName,
      appName: plan?.appName || '',
      description: plan?.description || '',
      appType: plan?.appType || '',
      design: plan?.design || '',
      completedStages: [],
      results: []
    };

    if (fs.existsSync(checkpointPath)) {
      try {
        const saved = JSON.parse(
          fs.readFileSync(checkpointPath, 'utf8')
        );

        if (
          saved &&
          (saved.version === 1 || saved.version === 2) &&
          saved.projectName === projectName &&
          Array.isArray(saved.completedStages)
        ) {
          checkpoint = saved;

          console.log(
            '[CHECKPOINT] تم استعادة حالة البناء السابقة'
          );

          console.log(
            '[CHECKPOINT] المراحل المكتملة: ' +
            checkpoint.completedStages.join(', ')
          );
        }
      } catch (error) {
        console.log(
          '[CHECKPOINT] الحالة السابقة غير صالحة — بدء جديد'
        );
      }
    }

    const results = Array.isArray(checkpoint.results)
      ? checkpoint.results
      : [];

    let existingFiles = collectDartContext(projectName);

    for (const stage of stages) {

      if (checkpoint.completedStages.includes(stage)) {
        console.log(
          `[CHECKPOINT] تخطي ${stage} — مكتملة مسبقاً`
        );

        existingFiles = collectDartContext(projectName);
        continue;
      }

      console.log(
        `[ENGINE] بدء مرحلة ${stage}`
      );

      try {
        const result = await generateStage(
          apiKey,
          plan,
          stage,
          existingFiles
        );

        const files = Array.isArray(result?.files)
          ? result.files
          : [];

        if (!files.length) {
          throw new Error(
            `AI لم يولد ملفات في مرحلة ${stage}`
          );
        }

        const written = writeFiles(
          projectName,
          files
        );

        if (!written) {
          throw new Error(
            `لم يتم كتابة أي ملف في مرحلة ${stage}`
          );
        }

        existingFiles = collectDartContext(
          projectName
        );

        results.push({
          stage,
          generated: files.length,
          written,
          files: existingFiles.length,
          completedAt: new Date().toISOString()
        });

        checkpoint.completedStages.push(stage);
        checkpoint.results = results;

        fs.writeFileSync(
          checkpointPath,
          JSON.stringify(checkpoint, null, 2),
          'utf8'
        );

        console.log(
          `[CHECKPOINT] تم حفظ ${stage} بنجاح`
        );

        console.log(
          `[ENGINE] اكتملت ${stage}: generated=${files.length}, written=${written}`
        );

      } catch (error) {

        checkpoint.results = results;

        fs.writeFileSync(
          checkpointPath,
          JSON.stringify(checkpoint, null, 2),
          'utf8'
        );

        console.log(
          `[CHECKPOINT] توقف البناء عند ${stage} — التقدم محفوظ`
        );

        throw error;
      }
    }

    if (
      checkpoint.completedStages.length === stages.length
    ) {
      try {
        fs.unlinkSync(checkpointPath);

        console.log(
          '[CHECKPOINT] اكتملت جميع مراحل AI — تم حذف checkpoint'
        );
      } catch (error) {
        console.log(
          '[CHECKPOINT] تعذر حذف checkpoint: ' +
          error.message
        );
      }
    }

console.log(
    '[ENGINE] التحقق من ملفات المشروع قبل GitHub Actions'
  );

  let validation =
    validateGeneratedFiles(
      projectName
    );

  if (!validation.valid) {
    throw new Error(
      `PROJECT_VALIDATION_FAILED: ${validation.errors.join('; ')}`
    );
  }

  console.log(
    '[ENGINE] تشغيل flutter pub get + analyze قبل GitHub'
  );

  try {
    execFileSync(
      'flutter',
      ['pub', 'get'],
      {
        cwd: projectRoot,
        stdio: 'inherit',
        timeout: 300000
      }
    );
  } catch (error) {
    throw new Error(
      'LOCAL_PUB_GET_FAILED: ' + error.message
    );
  }

  console.log(
    '[ENGINE] فحص Flutter الأولي'
  );

  let analyzePassed = false;

  try {
    execFileSync(
      'flutter',
      ['analyze', 'lib'],
      {
        cwd: projectRoot,
        stdio: 'inherit',
        timeout: 300000
      }
    );

    analyzePassed = true;
  } catch (error) {
    console.log(
      '[ENGINE] Flutter analyze اكتشف أخطاء — بدء الإصلاح الذكي'
    );
  }

  if (!analyzePassed) {
    await repairFlutterProject(
      apiKey,
      projectName,
      plan,
      3
    );
  }

  validation =
    validateGeneratedFiles(
      projectName
    );

  if (!validation.valid) {
    throw new Error(
      `PROJECT_VALIDATION_FAILED_AFTER_REPAIR: ${validation.errors.join('; ')}`
    );
  }

  console.log(
    '[ENGINE] المشروع اجتاز التحقق والإصلاح — جاهز لـ GitHub Actions'
  );

  writeGitHubWorkflow(projectName);

  const githubActions = require('./github-actions');

  const githubOwner = String(
    process.env.GITHUB_OWNER || ''
  ).trim();

  const githubToken = String(
    process.env.GITHUB_TOKEN || ''
  ).trim();

  if (!githubOwner) {
    throw new Error(
      'GITHUB_BUILD_FAILED: GITHUB_OWNER غير موجود في .env'
    );
  }

  if (!githubToken) {
    throw new Error(
      'GITHUB_BUILD_FAILED: GITHUB_TOKEN غير موجود في .env'
    );
  }

  const githubOutputDir = path.join(
    projectRoot,
    'github-build'
  );

  fs.mkdirSync(githubOutputDir, {
    recursive: true
  });

  console.log(
    '[ENGINE] بدء البناء الحقيقي عبر GitHub Actions'
  );

  const githubBuild =
    await githubActions.buildProject(
      githubOwner,
      projectName,
      githubToken,
      projectRoot,
      githubOutputDir
    );

  const apkPath = githubBuild.apkPath;

  console.log(
    '[ENGINE] تم إنشاء APK عبر GitHub Actions: ' +
    apkPath
  );

  return {
    projectRoot,
    apkPath,
    files: collectAllProjectFiles(
      projectName
    ),
    dartFiles: collectFiles(
      projectName
    ),
    importedPackages:
      validation.importedPackages,
    stages: results,
    validation
  };
}

const fs = require('fs');
const path = require('path');

function safeProjectName(name) {
  const raw = String(name || '').trim().toLowerCase();

  let safe = raw
    .replace(/[^a-z0-9_]+/g, '_')
    .replace(/^_+|_+$/g, '')
    .replace(/_+/g, '_');

  if (!safe) {
    safe = 'mahani_app_' + Date.now();
  }

  return safe;
}

function projectRoot(projectName) {
  return path.join(
    __dirname,
    'projects',
    safeProjectName(projectName)
  );
}

function ensureProject(projectName) {
  const root = projectRoot(projectName);

  fs.mkdirSync(root, { recursive: true });
  fs.mkdirSync(path.join(root, 'lib'), { recursive: true });
  fs.mkdirSync(path.join(root, '.github', 'workflows'), {
    recursive: true
  });

  return root;
}

function safeFilePath(filePath) {
  const value = String(filePath || '').trim();

  if (!value) return null;
  if (value.includes('..')) return null;
  if (value.includes('\\')) return null;
  if (!value.startsWith('lib/')) return null;
  if (!value.endsWith('.dart')) return null;

  return value;
}


function validateFilesPlan(files) {
  const input = Array.isArray(files) ? files : [];
  const errors = [];
  const seenPaths = new Map();
  const seenBasenames = new Map();

  for (const file of input) {
    if (!file || typeof file !== 'object') continue;

    const filePath = safeFilePath(file.path);

    if (!filePath) {
      errors.push({
        type: 'invalid_path',
        path: file.path || null
      });
      continue;
    }

    if (!String(file.content || '').trim()) {
      errors.push({
        type: 'empty_content',
        path: filePath
      });
      continue;
    }

    if (seenPaths.has(filePath)) {
      errors.push({
        type: 'duplicate_path',
        path: filePath,
        previous: seenPaths.get(filePath)
      });
    } else {
      seenPaths.set(filePath, filePath);
    }

    const basename = path.basename(filePath).toLowerCase();

    if (!seenBasenames.has(basename)) {
      seenBasenames.set(basename, []);
    }

    seenBasenames.get(basename).push(filePath);
  }

  for (const [basename, paths] of seenBasenames.entries()) {
    if (paths.length < 2) continue;

    const normalized = paths.map(p => p.toLowerCase());

    const conflictingArchitecture =
      normalized.some(p => p.includes('/core/models/')) &&
      normalized.some(p => p.includes('/models/'));

    const conflictingScreens =
      normalized.some(p => p.includes('/presentation/screens/')) &&
      normalized.some(p => p.includes('/screens/'));

    if (conflictingArchitecture || conflictingScreens) {
      errors.push({
        type: 'conflicting_duplicate_basename',
        basename,
        paths
      });
    }
  }

  return {
    valid: errors.length === 0,
    errors
  };
}


function normalizeDartImports(projectName, files) {
  const input = Array.isArray(files) ? files : [];
  const packageName = safeProjectName(projectName);

  return input.map(file => {
    if (!file || typeof file !== 'object') return file;

    const filePath = safeFilePath(file.path);
    if (!filePath) return file;

    let content = String(file.content || '');
    if (!content.trim()) return file;

    const regex = /import\s+(['"])([^'"]+)\1\s*;/g;

    content = content.replace(regex, (full, quote, importPath) => {
      if (
        importPath.startsWith('dart:') ||
        importPath.startsWith('package:')
      ) {
        return full;
      }

      if (
        importPath.endsWith('.dart') &&
        !importPath.startsWith('./') &&
        !importPath.startsWith('../') &&
        !importPath.startsWith('/')
      ) {
        const normalized = importPath.replace(/^\/+/, '');

        if (
          normalized.startsWith('core/') ||
          normalized.startsWith('domain/') ||
          normalized.startsWith('application/') ||
          normalized.startsWith('infrastructure/') ||
          normalized.startsWith('presentation/') ||
          normalized.startsWith('screens/') ||
          normalized.startsWith('services/') ||
          normalized.startsWith('models/') ||
          normalized.startsWith('widgets/') ||
          normalized.startsWith('theme/')
        ) {
          return "import " + quote +
            "package:" + packageName + "/" + normalized +
            quote + ";";
        }
      }

      return full;
    });

    return {
      ...file,
      content
    };
  });
}

function validateRelativeImportsPlan(files) {
  const input = Array.isArray(files) ? files : [];
  const errors = [];

  const knownFiles = new Set(
    input
      .map(file => safeFilePath(file?.path))
      .filter(Boolean)
      .map(filePath => filePath.replace(/\\/g, '/'))
  );

  for (const file of input) {
    if (!file || typeof file !== 'object') continue;

    const from = safeFilePath(file.path);
    if (!from) continue;

    const source = String(file.content || '');
    if (!source.trim()) continue;

    const regex =
      /import\s+['"]([^'"]+\.dart)['"]\s*;/g;

    let match;

    while ((match = regex.exec(source))) {
      const importPath = match[1];

      if (importPath.startsWith('package:')) {
        continue;
      }

      if (importPath.startsWith('dart:')) {
        continue;
      }

      const parts = from.split('/');
      parts.pop();

      if (importPath.startsWith('/')) {
        continue;
      }

      for (const part of importPath.split('/')) {
        if (!part || part === '.') continue;

        if (part === '..') {
          if (parts.length > 1) {
            parts.pop();
          }
          continue;
        }

        parts.push(part);
      }

      const resolved = parts.join('/');

      if (!knownFiles.has(resolved)) {
        errors.push({
          type: 'missing_relative_import',
          from,
          import: importPath,
          resolved
        });
      }
    }
  }

  return errors;
}

function validatePackageImportsPlan(projectName, files) {
  const input = Array.isArray(files) ? files : [];
  const errors = [];
  const projectPackage = safeProjectName(projectName);

  const knownPackages = new Set([
    'flutter',
    'flutter_test',
    'cupertino_icons',
    'provider',
    'sqflite',
    'path',
    'shared_preferences',
    'intl',
    'flutter_local_notifications',
    'pdf',
    'printing',
    'excel',
    'file_picker',
    'permission_handler',
    'path_provider',
    'http',
    'uuid',
    'crypto',
    projectPackage
  ]);

  for (const file of input) {
    if (!file || typeof file !== 'object') continue;

    const filePath = safeFilePath(file.path);
    if (!filePath) continue;

    const source = String(file.content || '');

    const regex = /import\s+['"]package:([^/]+)\/[^'"]+['"]\s*;/g;

    let match;

    while ((match = regex.exec(source))) {
      const packageName = match[1];

      if (!knownPackages.has(packageName)) {
        errors.push({
          type: 'unknown_package_import',
          path: filePath,
          package: packageName
        });
      }
    }
  }

  return errors;
}

function writeFiles(projectName, files) {
  files = normalizeDartImports(projectName, files);

  const validation = validateFilesPlan(files);

  const importErrors = [
    ...validateRelativeImportsPlan(files),
    ...validatePackageImportsPlan(projectName, files)
  ];

  if (!validation.valid || importErrors.length > 0) {
    const error = new Error('INVALID_FILES_PLAN');
    error.code = 'INVALID_FILES_PLAN';
    error.details = [
      ...validation.errors,
      ...importErrors
    ];
    throw error;
  }

  const root = ensureProject(projectName);

  let written = 0;

  for (const file of Array.isArray(files) ? files : []) {
    if (!file || typeof file !== 'object') continue;

    const filePath = safeFilePath(file.path);
    if (!filePath) continue;

    const content = String(file.content || '');
    if (!content.trim()) continue;

    const fullPath = path.resolve(root, filePath);
    const libRoot = path.resolve(root, 'lib');

    if (!fullPath.startsWith(libRoot + path.sep)) {
      continue;
    }

    fs.mkdirSync(path.dirname(fullPath), {
      recursive: true
    });

    fs.writeFileSync(
      fullPath,
      content.endsWith('\n') ? content : content + '\n',
      'utf8'
    );

    written++;
  }

  return {
    root,
    written,
    validation: {
      files: validation,
      importErrors
    }
  };
}

function listDartFiles(projectName) {
  const root = projectRoot(projectName);
  const libRoot = path.join(root, 'lib');

  if (!fs.existsSync(libRoot)) {
    return [];
  }

  const result = [];

  function walk(dir) {
    for (const entry of fs.readdirSync(dir, {
      withFileTypes: true
    })) {
      const full = path.join(dir, entry.name);

      if (entry.isDirectory()) {
        walk(full);
        continue;
      }

      if (entry.isFile() && entry.name.endsWith('.dart')) {
        result.push(path.relative(root, full));
      }
    }
  }

  walk(libRoot);

  return result.sort();
}

function findMissingRelativeImports(projectName) {
  const root = projectRoot(projectName);
  const dartFiles = listDartFiles(projectName);
  const missing = [];

  for (const relativeFile of dartFiles) {
    const fullFile = path.join(root, relativeFile);
    const source = fs.readFileSync(fullFile, 'utf8');

    const regex =
      /import\s+['"](\.{1,2}\/[^'"]+\.dart)['"]\s*;/g;

    let match;

    while ((match = regex.exec(source))) {
      const target = path.resolve(
        path.dirname(fullFile),
        match[1]
      );

      if (!fs.existsSync(target)) {
        missing.push({
          from: relativeFile,
          import: match[1]
        });
      }
    }
  }

  return missing;
}


function findDuplicateDartBasenames(projectName) {
  const files = listDartFiles(projectName);
  const groups = new Map();

  for (const file of files) {
    const basename = path.basename(file).toLowerCase();

    if (!groups.has(basename)) {
      groups.set(basename, []);
    }

    groups.get(basename).push(file);
  }

  const duplicates = [];

  for (const [basename, paths] of groups.entries()) {
    if (paths.length > 1) {
      duplicates.push({
        basename,
        paths
      });
    }
  }

  return duplicates;
}

function scanForbiddenPatterns(projectName) {
  const root = projectRoot(projectName);
  const dartFiles = listDartFiles(projectName);

  const patterns = [
    /Future\.delayed/g,
    /\bTODO\b/g,
    /\bplaceholder\b/gi,
    /\bexample\b/gi,
    /تم تحليل البيانات/g,
    /بيانات وهمية/g
  ];

  const findings = [];

  for (const relativeFile of dartFiles) {
    const fullFile = path.join(root, relativeFile);
    const source = fs.readFileSync(fullFile, 'utf8');

    for (const pattern of patterns) {
      if (pattern.test(source)) {
        findings.push({
          file: relativeFile,
          pattern: pattern.toString()
        });

        pattern.lastIndex = 0;
      }
    }
  }

  return findings;
}

function getProjectReport(projectName) {
  return {
    project: safeProjectName(projectName),
    files: listDartFiles(projectName),
    missingImports: findMissingRelativeImports(projectName),
    duplicateFiles: findDuplicateDartBasenames(projectName),
    forbiddenPatterns: scanForbiddenPatterns(projectName)
  };
}

function classifyFile(filePath) {
  const p = String(filePath || '');

  if (
    p.includes('/core/') ||
    p.includes('/domain/') ||
    p.includes('/models/') ||
    p.endsWith('/entities.dart')
  ) {
    return 'core';
  }

  if (
    p.includes('/database/') ||
    p.includes('/repositories/')
  ) {
    return 'data';
  }

  if (
    p.includes('/services/') ||
    p.includes('/infrastructure/')
  ) {
    return 'services';
  }

  if (
    p.includes('/screens/') ||
    p.includes('/widgets/') ||
    p.includes('/presentation/') ||
    p.includes('/providers/')
  ) {
    return 'ui';
  }

  if (p === 'lib/main.dart') {
    return 'core';
  }

  return 'core';
}

function filesForStage(plan, stage) {
  const files = Array.isArray(plan?.files) ? plan.files : [];

  return files.filter(file => {
    if (!file || typeof file !== 'object') return false;
    if (!file.path || !file.content) return false;

    return classifyFile(file.path) === stage;
  });
}

function executeStage(projectName, stage, plan) {
  if (!stageManager.isValidStage(stage)) {
    throw new Error(`Invalid stage: ${stage}`);
  }

  const root = rootFor(projectName);

  if (stage === 'plan') {
    return {
      stage,
      root,
      written: 0,
      message: 'Plan validated'
    };
  }

  if (stage === 'core') {
    writeBaseFiles(projectName, plan);

    const result = staged.writeFiles(
      projectName,
      filesForStage(plan, 'core')
    );

    return {
      stage,
      root,
      written: result.written,
      message: 'Core stage completed'
    };
  }

  if (stage === 'data') {
    const result = staged.writeFiles(
      projectName,
      filesForStage(plan, 'data')
    );

    return {
      stage,
      root,
      written: result.written,
      message: 'Data stage completed'
    };
  }

  if (stage === 'services') {
    const result = staged.writeFiles(
      projectName,
      filesForStage(plan, 'services')
    );

    return {
      stage,
      root,
      written: result.written,
      message: 'Services stage completed'
    };
  }

  if (stage === 'ui') {
    const result = staged.writeFiles(
      projectName,
      filesForStage(plan, 'ui')
    );

    return {
      stage,
      root,
      written: result.written,
      message: 'UI stage completed'
    };
  }

  if (stage === 'verify') {
    const report = staged.getProjectReport(projectName);

    return {
      stage,
      root,
      written: 0,
      report,
      valid:
      report.files.length > 0 &&
      report.missingImports.length === 0 &&
      (!report.duplicateFiles || report.duplicateFiles.length === 0) &&
      report.forbiddenPatterns.length === 0,
    message: 'Verification completed'
    };
  }

  if (stage === 'repair') {
    let report = staged.getProjectReport(projectName);
    let repaired = 0;
    const repairedFiles = [];

    /*
     * Repair is intentionally conservative.
     * We only modify files that contain a directly detectable
     * forbidden pattern. Missing imports are reported but are
     * not guessed automatically.
     */

    const rootPath = staged.projectRoot(projectName);

    for (const finding of report.forbiddenPatterns) {
      if (!finding || !finding.file) continue;

      const filePath = path.join(rootPath, finding.file);

      if (!fs.existsSync(filePath)) continue;

      let source = fs.readFileSync(filePath, 'utf8');
      const before = source;

      source = source
        .split('Future.delayed').join('')
        .split('TODO').join('')
        .split('todo').join('')
        .split('Todo').join('')
        .split('placeholder').join('')
        .split('Placeholder').join('')
        .split('PLACEHOLDER').join('')
        .split('example').join('')
        .split('Example').join('')
        .split('EXAMPLE').join('')
        .split('تم تحليل البيانات').join('')
        .split('بيانات وهمية').join('');

      if (source !== before) {
        fs.writeFileSync(filePath, source, 'utf8');
        repaired++;
        repairedFiles.push(finding.file);
      }
    }

    report = staged.getProjectReport(projectName);

    const valid =
      report.files.length > 0 &&
      report.missingImports.length === 0 &&
      (!report.duplicateFiles || report.duplicateFiles.length === 0) &&
      report.forbiddenPatterns.length === 0;

    return {
      stage,
      root,
      written: repaired,
      repaired: repaired > 0,
      repairedFiles,
      report,
      valid,
      message: valid
        ? 'Repair completed and verification passed'
        : 'Repair completed but project still requires fixes'
    };
  }

  throw new Error(`Unsupported stage: ${stage}`);
}

function runAll(projectName, plan) {
  let state = stageManager.loadOrCreateState(projectName);
  const results = [];

  for (const stage of stageManager.STAGES) {
    try {
      const result = executeStage(projectName, stage, plan);
      results.push(result);
      state = stageManager.completeStage(state, stage);
    } catch (error) {
      state = stageManager.failStage(state, stage, error.message);
      throw error;
    }
  }

  return {
    state,
    results,
    report: staged.getProjectReport(projectName)
  };
}

const stateStore = require('./stage-state');

const STAGES = Object.freeze([
  'plan',
  'core',
  'data',
  'services',
  'ui',
  'verify',
  'repair'
]);

function isValidStage(stage) {
  return STAGES.includes(stage);
}

function nextStage(stage) {
  const index = STAGES.indexOf(stage);

  if (index === -1 || index >= STAGES.length - 1) {
    return null;
  }

  return STAGES[index + 1];
}

function previousStage(stage) {
  const index = STAGES.indexOf(stage);

  if (index <= 0) {
    return null;
  }

  return STAGES[index - 1];
}

function createState(projectName) {
  const state = {
    projectName,
    currentStage: 'plan',
    completed: [],
    failed: [],
    startedAt: new Date().toISOString(),
    updatedAt: new Date().toISOString()
  };

  stateStore.saveState(projectName, state);

  return state;
}

function loadState(projectName) {
  return stateStore.loadState(projectName);
}

function loadOrCreateState(projectName) {
  const existing = loadState(projectName);

  if (existing) {
    return existing;
  }

  return createState(projectName);
}

function saveState(state) {
  if (!state || !state.projectName) {
    throw new Error('Invalid stage state');
  }

  state.updatedAt = new Date().toISOString();

  stateStore.saveState(
    state.projectName,
    state
  );

  return state;
}

function completeStage(state, stage) {
  if (!isValidStage(stage)) {
    throw new Error(`Invalid stage: ${stage}`);
  }

  if (!state.completed.includes(stage)) {
    state.completed.push(stage);
  }

  state.failed = state.failed.filter(
    item => item.stage !== stage
  );

  state.currentStage =
    nextStage(stage) || stage;

  saveState(state);

  return state;
}

function failStage(state, stage, error) {
  if (!isValidStage(stage)) {
    throw new Error(`Invalid stage: ${stage}`);
  }

  state.failed.push({
    stage,
    error: String(error || 'Unknown error'),
    at: new Date().toISOString()
  });

  state.currentStage = stage;

  saveState(state);

  return state;
}

function resetState(projectName) {
  return createState(projectName);
}

module.exports = {
  STAGES,
  isValidStage,
  nextStage,
  previousStage,
  createState,
  loadState,
  loadOrCreateState,
  saveState,
  completeStage,
  failStage,
  resetState
};

function safeName(name) {
  return String(name || 'mahani_app')
    .trim()
    .toLowerCase()
    .replace(/[^a-z0-9_]+/g, '_')
    .replace(/^_+|_+$/g, '')
    .replace(/_+/g, '_') || 'mahani_app';

function statePath(projectName) {
  return path.join(
    __dirname,
    'projects',
    safeName(projectName),
    '.mahani-state.json'
  );
}

function saveState(projectName, state) {
  const file = statePath(projectName);

  fs.mkdirSync(path.dirname(file), {
    recursive: true
  });

  fs.writeFileSync(
    file,
    JSON.stringify(state, null, 2),
    'utf8'
  );

  return file;
}

function loadState(projectName) {
  const file = statePath(projectName);

  if (!fs.existsSync(file)) {
    return null;
  }

  return JSON.parse(
    fs.readFileSync(file, 'utf8')
  );
}

function deleteState(projectName) {
  const file = statePath(projectName);

  if (fs.existsSync(file)) {
    fs.unlinkSync(file);
    return true;
  }

  return false;
}

function stateExists(projectName) {
  return fs.existsSync(statePath(projectName));
}

function githubRequest(method, apiPath, token, body = null) {
  return new Promise((resolve, reject) => {
    const req = https.request({
      hostname: 'api.github.com',
      path: apiPath,
      method,
      headers: {
        Accept: 'application/vnd.github+json',
        Authorization: `Bearer ${token}`,
        'X-GitHub-Api-Version': '2022-11-28',
        'User-Agent': 'Mahani-Builder-AI',
        ...(body ? {
          'Content-Type': 'application/json',
          'Content-Length': Buffer.byteLength(body)
        } : {})
      }
    }, res => {
      let data = '';

      res.on('data', chunk => {
        data += chunk;
      });

      res.on('end', () => {
        let parsed = {};

        try {
          parsed = data ? JSON.parse(data) : {};
        } catch {
          return reject(
            new Error(`GitHub returned invalid JSON: HTTP ${res.statusCode}`)
          );
        }

        if (res.statusCode < 200 || res.statusCode >= 300) {
          const error = new Error(`GitHub API HTTP ${res.statusCode}`);
          error.status = res.statusCode;
          error.data = parsed;
          return reject(error);
        }

        resolve(parsed);
      });
    });

    req.on('error', reject);

    if (body) req.write(body);

    req.end();
  });
}

function runCommand(command, args, options = {}) {
  return new Promise((resolve, reject) => {
    execFile(command, args, {
      cwd: options.cwd,
      env: {
        ...process.env,
        ...(options.env || {})
      },
      maxBuffer: 20 * 1024 * 1024
    }, (error, stdout, stderr) => {
      if (error) {
        error.stdout = stdout;
        error.stderr = stderr;
        return reject(error);
      }

      resolve({
        stdout: stdout || '',
        stderr: stderr || ''
      });
    });
  });
}

function normalizeRepoName(name) {
  let value = String(name || 'mahani-app')
    .trim()
    .toLowerCase()
    .replace(/[^a-z0-9._-]+/g, '-')
    .replace(/^-+|-+$/g, '');

  if (!value) value = `mahani-app-${Date.now()}`;

  if (value.length > 90) {
    value = value.slice(0, 90);
  }

  return value;
}

async function getRepository(owner, repo, token) {
  try {
    return await githubRequest(
      'GET',
      `/repos/${encodeURIComponent(owner)}/${encodeURIComponent(repo)}`,
      token
    );
  } catch (error) {
    if (error.status === 404) return null;
    throw error;
  }
}

async function ensureRepository(owner, repo, token) {
  const existing = await getRepository(owner, repo, token);

  if (existing) {
    return existing;
  }

  console.log(`[GITHUB] Creating repository ${owner}/${repo}`);

  return githubRequest(
    'POST',
    '/user/repos',
    token,
    JSON.stringify({
      name: repo,
      description: 'Generated by Mahani Builder AI',
      private: false,
      has_issues: true,
      has_projects: false,
      has_wiki: false,
      auto_init: false
    })
  );
}

async function pushProjectToGitHub(owner, repo, token, projectPath) {
  if (!fs.existsSync(projectPath)) {
    throw new Error(`Project path not found: ${projectPath}`);
  }

  console.log(`[GITHUB] Preparing project ${projectPath}`);

  const gitDir = path.join(projectPath, '.git');

  if (fs.existsSync(gitDir)) {
    await runCommand('git', ['remote', 'remove', 'origin'], {
      cwd: projectPath
    }).catch(() => {});
  } else {
    await runCommand('git', ['init', '-b', 'main'], {
      cwd: projectPath
    });
  }

  const remoteUrl =
    `https://x-access-token:${encodeURIComponent(token)}` +
    `@github.com/${owner}/${repo}.git`;

  await runCommand('git', [
    'remote',
    'add',
    'origin',
    remoteUrl
  ], {
    cwd: projectPath
  });

  await runCommand('git', [
    'config',
    'user.name',
    'Mahani Builder AI'
  ], {
    cwd: projectPath
  });

  await runCommand('git', [
    'config',
    'user.email',
    'mahani-builder@users.noreply.github.com'
  ], {
    cwd: projectPath
  });

  await runCommand('git', ['add', '.'], {
    cwd: projectPath
  });

  await runCommand('git', [
    'commit',
    '-m',
    'feat: generate project with Mahani Builder AI'
  ], {
    cwd: projectPath
  }).catch(async error => {
    if (
      !String(error.stdout || '').includes('nothing to commit') &&
      !String(error.stderr || '').includes('nothing to commit')
    ) {
      throw error;
    }
  });

  await runCommand('git', [
    'push',
    '-u',
    'origin',
    'main',
    '--force'
  ], {
    cwd: projectPath
  });

  const shaResult = await runCommand('git', [
    'rev-parse',
    'HEAD'
  ], {
    cwd: projectPath
  });

  const commitSha = String(shaResult.stdout || '').trim();

  if (!commitSha) {
    throw new Error('لم يتم الحصول على commit SHA بعد push');
  }

  console.log(`[GITHUB] Project pushed to ${owner}/${repo}`);
  console.log(`[GITHUB] Commit SHA: ${commitSha}`);

  return {
    url: `https://github.com/${owner}/${repo}`,
    commitSha
  };
}

async function waitForWorkflowReady(owner, repo, workflowFile, token, timeout = 60000) {
  const started = Date.now();

  while (Date.now() - started < timeout) {
    try {
      const data = await githubRequest(
        'GET',
        `/repos/${encodeURIComponent(owner)}/${encodeURIComponent(repo)}/actions/workflows`,
        token
      );

      const workflows = data.workflow_runs || data.workflows || [];

      const workflow = workflows.find(item =>
        item.path === workflowFile ||
        item.name === workflowFile ||
        item.html_url?.endsWith(workflowFile)
      );

      if (workflow && workflow.state === 'active') {
        console.log(`[GITHUB] Workflow ready: ${workflow.id}`);
        return workflow;
      }

      console.log('[GITHUB] Waiting for workflow registration...');
    } catch (error) {
      console.log('[GITHUB] Waiting for workflow API...');
    }

    await new Promise(resolve => setTimeout(resolve, 3000));
  }

  throw new Error('GITHUB_WORKFLOW_NOT_READY: workflow was not registered in time');
}

async function dispatchWorkflow(owner, repo, workflowFile, ref, token) {
  await githubRequest(

    `/repos/${encodeURIComponent(owner)}/${encodeURIComponent(repo)}` +
    `/actions/workflows/${encodeURIComponent(workflowFile)}/dispatches`,

    JSON.stringify({ ref })
  );
}

async function getWorkflowRun(owner, repo, runId, token) {
  return githubRequest(
    'GET',
    `/repos/${encodeURIComponent(owner)}/${encodeURIComponent(repo)}` +

async function waitForNewWorkflowRun(
  owner,
  repo,
  token,
  previousIds = []
) {
  for (let i = 0; i < 40; i++) {
    const data = await getWorkflowRuns(owner, repo, token);

    const run = (data.workflow_runs || []).find(
      item => !previousIds.includes(item.id)
    );

    if (run) return run;

    await new Promise(resolve => setTimeout(resolve, 3000));
  }

  throw new Error('لم يتم العثور على تشغيل GitHub Actions الجديد');
}

async function waitForWorkflowRunBySha(
  owner,
  repo,
  token,
  commitSha,
  timeout = 120000
) {
  const started = Date.now();

  while (Date.now() - started < timeout) {
    const data = await getWorkflowRuns(
      owner,
      repo,
      token
    );

    const runs = data.workflow_runs || [];

    const run = runs.find(item =>
      item.head_sha === commitSha &&
      item.event === 'push'
    );

    if (run) {
      console.log(`[GITHUB] Workflow run found: ${run.id}`);
      console.log(`[GITHUB] Run SHA: ${run.head_sha}`);
      console.log(`[GITHUB] Run status: ${run.status}`);
      return run;
    }

    await new Promise(resolve =>
      setTimeout(resolve, 3000)
    );
  }

  throw new Error(
    `لم يتم العثور على GitHub Actions Run للـ commit ${commitSha}`
  );
}

async function waitForWorkflowSuccess(
  owner,
  repo,
  runId,
  token,
  timeout = 20 * 60 * 1000
) {
  const started = Date.now();

  while (Date.now() - started < timeout) {
    const run = await getWorkflowRun(
      owner,
      repo,
      runId,
      token
    );

    console.log(
      `[GITHUB] Run ${run.id}: status=${run.status}, conclusion=${run.conclusion}`
    );

    if (run.status === 'completed') {
      if (run.conclusion !== 'success') {
        throw new Error(
          `GitHub Actions failed: ${run.conclusion || 'unknown'}`
        );
      }

      return run;
    }

    await new Promise(resolve => setTimeout(resolve, 10000));
  }

  throw new Error('GitHub Actions build timed out');
}

function downloadUrl(url, token) {
  return new Promise((resolve, reject) => {
    const requestUrl = new URL(url);

    const headers = {
      Accept: 'application/vnd.github+json',
      'User-Agent': 'Mahani-Builder-AI'
    };

    if (requestUrl.hostname === 'api.github.com') {
      headers.Authorization = `Bearer ${token}`;
      headers['X-GitHub-Api-Version'] = '2022-11-28';
    }

    const req = https.request({
      hostname: requestUrl.hostname,
      port: requestUrl.port || 443,
      path: requestUrl.pathname + requestUrl.search,
      method: 'GET',
      headers
    }, res => {
      if (
        res.statusCode >= 300 &&
        res.statusCode < 400 &&
        res.headers.location
      ) {
        return resolve(
          downloadUrl(res.headers.location, token)
        );
      }

      const chunks = [];

      res.on('data', chunk => chunks.push(chunk));

      res.on('end', () => {
        if (res.statusCode < 200 || res.statusCode >= 300) {
          return reject(
            new Error(
              `GitHub download failed: HTTP ${res.statusCode}`
            )
          );
        }

        resolve(Buffer.concat(chunks));
      });
    });

    req.on('error', reject);
    req.end();
  });
}

async function downloadArtifact(
  owner,
  repo,
  artifactId,
  token,
  outputZip
) {
  const url =
    `https://api.github.com/repos/${encodeURIComponent(owner)}` +
    `/${encodeURIComponent(repo)}` +
    `/actions/artifacts/${artifactId}/zip`;

  const buffer = await downloadUrl(url, token);

  fs.mkdirSync(path.dirname(outputZip), {
    recursive: true
  });

  fs.writeFileSync(outputZip, buffer);

  return outputZip;
}

async function extractApk(zipPath, outputDir) {
  fs.mkdirSync(outputDir, {
    recursive: true
  });

  await runCommand('unzip', [
    '-o',
    zipPath,
    '-d',
    outputDir
  ]);

  const apkPath = path.join(
    outputDir,
    'app-release.apk'
  );

  if (!fs.existsSync(apkPath)) {
    throw new Error(
      'app-release.apk not found inside GitHub artifact'
    );
  }

  return apkPath;
}

async function buildProject(
  owner,
  repo,
  token,
  projectPath,
  outputDir
) {
  const repository = await ensureRepository(
    owner,
    repo,
    token
  );

  console.log('[GITHUB] Pushing project to main...');

  const pushResult = await pushProjectToGitHub(
    owner,
    repo,
    token,
    projectPath
  );

  const repoUrl = pushResult.url;
  const commitSha = pushResult.commitSha;

  console.log(`[GITHUB] Waiting for workflow for commit ${commitSha}...`);

  const run = await waitForWorkflowRunBySha(
    owner,
    repo,
    token,
    commitSha
  );

  console.log(`[GITHUB] New workflow run: ${run.id}`);

  const completed = await waitForWorkflowSuccess(
    owner,
    repo,
    run.id,
    token
  );

  const artifacts = await getRunArtifacts(
    owner,
    repo,
    completed.id,
    token
  );

  const artifact =
    (artifacts.artifacts || []).find(
      item =>
        item.name === 'mahani_app-apk' &&
        item.expired === false
    ) ||
    (artifacts.artifacts || []).find(
      item =>
        String(item.name).toLowerCase().includes('apk') &&
        item.expired === false
    );

  if (!artifact) {
    throw new Error(
      'GitHub Actions succeeded but no APK artifact was found'
    );
  }

  const zipPath = path.join(
    outputDir,
    `artifact-${artifact.id}.zip`
  );

  await downloadArtifact(
    owner,
    repo,
    artifact.id,
    token,
    zipPath
  );

  const apkPath = await extractApk(
    zipPath,
    outputDir
  );

  return {
    repository: {
      name: repository.name,
      fullName: repository.full_name,
      url: repoUrl
    },
    runId: completed.id,
    workflowUrl: completed.html_url,
    artifactId: artifact.id,
    artifactName: artifact.name,
    apkPath,
    apkSize: fs.statSync(apkPath).size
  };
}
